var SHEET_NAME = "contest"; function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = rows[i][j]; } data.push(obj); } return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var params = JSON.parse(e.postData.contents); var action = params.action; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (action === "add") { var newId = new Date().getTime(); sheet.appendRow([newId, params.title, params.author, params.url, 0]); } else if (action === "vote") { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] == params.id) { var currentVotes = Number(rows[i][4]) || 0; sheet.getRange(i + 1, 5).setValue(currentVotes + 1); break; } } } return ContentService.createTextOutput(JSON.stringify({result: "success"})) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
